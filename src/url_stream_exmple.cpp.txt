#include "AudioTools.h"
#include "AudioTools/AudioCodecs/CodecMP3Helix.h"

#define DAC_PIN_LRC 13
#define DAC_PIN_BCLK 12
#define DAC_PIN_DIN 14


URLStream url("MISINFRA", "123456#@227");
I2SStream i2s; // final output of decoded stream
VolumeStream volume(i2s);
EncodedAudioStream dec(&volume, new MP3DecoderHelix()); // Decoding stream
StreamCopy copier(dec, url);                            // copy url to decoder

void setup()
{
  Serial.begin(115200);
  Serial.println("Started\n\n\n\n\n\n\n\n\n\n");
  delay(2000);
  while (!Serial)
    Serial.print(".");
  Serial.println();
  Serial.println("Serial Started");
  AudioLogger::instance().begin(Serial, AudioLogger::Info);
  // setup i2s
  auto config = i2s.defaultConfig(TX_MODE);
  Serial.println("1\n\n\n...");
  config.pin_bck = DAC_PIN_BCLK;
  Serial.println("2\n\n\n...");
  config.pin_ws = DAC_PIN_LRC;
  Serial.println("3\n\n\n...");
  config.pin_data = DAC_PIN_DIN;
  Serial.println("4\n\n\n...");

  // config.sample_rate = 22050; // Match the MP3 sample rate
  // config.bits_per_sample = 16;
  // config.channels = 1;                // Mono
  // config.i2s_format = I2S_STD_FORMAT; // Standard I2S format
  // config.port_no = 0;                 // I2S port number (usually 0)

  i2s.begin(config);
  Serial.println("5\n\n\n...");
  Serial.println("After i2s begin");

  Serial.println("Before dec begin");
  // setup I2S based on sampling rate provided by decoder
  dec.begin();
  Serial.println("After dec begin");

  // set initial volume
  auto vcfg = volume.defaultConfig();
  vcfg.copyFrom(config);
  // vcfg.allow_boost = true; // activate amplification using linear control
  volume.begin(vcfg); // we need to provide the bits_per_sample and channels
  volume.setVolume(0.3);
  Serial.println("Volume initialized");

  // mp3 radio
  url.begin("http://192.168.21.44/received_audio.mp3", "audio/mp3");
  Serial.println("After url begin");
}

void loop()
{
  size_t bytes_copied = copier.copy();
  if (bytes_copied > 0)
  {
    Serial.printf("Bytes copied:%d\n", bytes_copied);
  }
  else
  {
    Serial.printf("Copy ended\n");
    delay(1000);
  }
}
