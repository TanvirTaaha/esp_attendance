#include "attendance.h"

AudioInfo info(44100, 2, 16);
SineWaveGenerator<int16_t> sineWave(32);       // subclass of SoundGenerator with max amplitude of 32000
GeneratedSoundStream<int16_t> sound(sineWave); // Stream generated from sine wave
I2SStream i2sStream;
VolumeStream volumeStream(i2sStream);
StreamCopy copier(volumeStream, sound);

void sinewave_generator_init()
{

  // AudioToolsLogger.begin(Serial, AudioToolsLogLevel::Info);

  // start I2S
  Serial.println("starting I2S...");
  auto config = i2sStream.defaultConfig(TX_MODE);
  config.copyFrom(info);
  i2sStream.begin(config);

  auto vcfg = volumeStream.defaultConfig();
  vcfg.copyFrom(config);
  volumeStream.begin(vcfg);
  volumeStream.setVolume(0.01f);
  // Setup sine wave
  sineWave.begin(info, N_B4);
  Serial.println("started...");
}

void sinewave_generator_loop()
{
  copier.copy();
}
