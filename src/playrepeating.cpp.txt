float vol;
char cmd;
uint8_t *buffer_mp3;
uint8_t *buffer_decoded;
MemoryStream demo_audio_data(buffer_decoded, demo_base64_himel_size);
void setup()
{
  Serial.begin(115200);
  delay(2000);
  buffer_decoded = (uint8_t *)malloc(demo_base64_himel_size);
  audio_init();
  AudioToolsLogger.begin(Serial, AudioToolsLogLevel::Info);
}

void loop()
{
  restart_audio();
  int padding = 0;
  while (demo_base64_himel[demo_base64_himel_size - padding - 1] == '=')
  {
    padding++;
  }
  size_t expected_decoded_len = (demo_base64_himel_size / 4) * 3 - padding;
  size_t decoded_bytes_written;
  Serial.printf("expected_len:%d, decoded_bytes_written:%d\n", expected_decoded_len, decoded_bytes_written);
  auto ret = mbedtls_base64_decode((uint8_t *)buffer_decoded, expected_decoded_len, &decoded_bytes_written, (uint8_t *)demo_base64_himel, demo_base64_himel_size);
  if (ret == 0)
  {
    Serial.println("Success");
    Serial.printf("expected_len:%d, decoded_bytes_written:%d\n", expected_decoded_len, decoded_bytes_written);
    demo_audio_data.setValue((uint8_t *)buffer_decoded, decoded_bytes_written);
    demo_audio_data.resize(decoded_bytes_written);
    restart_audio();
    Serial.println("demo_audio_data set value done");
  }
  else
  {
    Serial.printf("failed, ret:%d\n", ret);
  }

  Serial.printf("copier.available:%d\naudio_data.available:%d\n", copier.available(), demo_audio_data.available());
  while (copier.available() > 0)
  {
    copier.copy();
    Serial.println("copying..");
  }

  free(buffer_decoded);
  delay(1000);
}
